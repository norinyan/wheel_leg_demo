cmake_minimum_required(VERSION 3.8)
project(wheel_leg_demo)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(std_msgs REQUIRED)
find_package(std_srvs REQUIRED)
find_package(gazebo_ros REQUIRED)
find_package(gazebo_msgs REQUIRED)
find_package(nav_msgs REQUIRED)
find_package(visualization_msgs REQUIRED)
find_package(PkgConfig REQUIRED)
pkg_check_modules(GAZEBO REQUIRED IMPORTED_TARGET gazebo)
find_package(TBB REQUIRED)
find_package(Ceres REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(trajectory_msgs REQUIRED)
add_library(wheel_leg_animation SHARED src/demo_node.cpp)
target_include_directories(wheel_leg_animation PRIVATE include)
target_compile_features(wheel_leg_animation PRIVATE cxx_std_17)
target_compile_options(wheel_leg_animation PRIVATE -Wall -Wextra)
target_link_libraries(wheel_leg_animation PkgConfig::GAZEBO TBB::tbb)
ament_target_dependencies(wheel_leg_animation rclcpp sensor_msgs std_msgs std_srvs gazebo_ros gazebo_msgs nav_msgs geometry_msgs trajectory_msgs)
add_executable(global_planner src/global_planner.cpp src/wheel_local.cpp src/teb_optimizer.cpp src/foot_local.cpp src/task_runner.cpp)
target_include_directories(global_planner PRIVATE include)
target_compile_features(global_planner PRIVATE cxx_std_17)
target_compile_options(global_planner PRIVATE -O2 -Wall -Wextra)
target_link_libraries(global_planner PkgConfig::GAZEBO TBB::tbb Ceres::ceres)
ament_target_dependencies(global_planner rclcpp nav_msgs visualization_msgs gazebo_msgs std_msgs std_srvs geometry_msgs trajectory_msgs)
install(TARGETS global_planner DESTINATION lib/${PROJECT_NAME})
install(TARGETS wheel_leg_animation LIBRARY DESTINATION lib)
install(DIRECTORY urdf worlds config launch DESTINATION share/${PROJECT_NAME})
if(BUILD_TESTING)
  enable_testing()
  add_executable(foot_planner_check test/foot_planner_check.cpp)
  target_include_directories(foot_planner_check PRIVATE include)
  target_compile_features(foot_planner_check PRIVATE cxx_std_17)
  target_compile_options(foot_planner_check PRIVATE -O2 -Wall -Wextra)
  add_test(NAME foot_planner_check COMMAND foot_planner_check)
endif()
ament_package()
